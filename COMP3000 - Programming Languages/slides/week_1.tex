\begin{frame}{How the classes run -- Team Based Learning}
\phantomsection\label{how-the-classes-run-team-based-learning}
\begin{itemize}
\tightlist
\item
  The unit runs on \textbf{TBL}: you work in a team for all 13 weeks
\item
  Each week follows a fixed cycle:

  \begin{itemize}
  \tightlist
  \item
    \textbf{iRAT} -- individual multiple-choice quiz (done before class)
  \item

  \item

  \item

\note{iRAT before class; tRAT + FAT + application task in class. Team

\begin{frame}{Domain-specific languages -- the big idea}
\phantomsection\label{domain-specific-languages-the-big-idea}
\begin{itemize}
\tightlist
\item
  A \textbf{domain-specific language} (DSL) is a language built for one
  job
\item
  Ch 1 calls them \emph{``little languages''} -- ``pidgins tailor-built
  to a specific task''
\item
  Think application scripting languages, template engines, markup
  formats, config files
\item
  Contrast with a \textbf{general-purpose language} (GPL): Java, C,
  Python -- built for any job
\end{itemize}

\note{Definition from the textbook: ``For every successful
general-purpose language, there are a thousand successful niche ones.''}
\end{frame}

\begin{frame}{Little languages are everywhere}
\phantomsection\label{little-languages-are-everywhere}
\includegraphics[width=0.5\textwidth]{text/site/image/introduction/little-languages.png}

\begin{itemize}
\tightlist
\item
  SQL, regex, LaTeX, Make, CSS, shell -- all little languages
\item
  Almost every large software project needs a handful
\item
  When you can, \textbf{reuse an existing one} instead of rolling your
  own
\end{itemize}

\note{``Once you factor in documentation, debuggers, editor support,
syntax highlighting, and all the other trappings, doing it yourself
becomes a tall order.'' -- ch 1}
\end{frame}

\begin{frame}{Why build a DSL at all?}
\phantomsection\label{why-build-a-dsl-at-all}
\begin{itemize}
\tightlist
\item
  When a general-purpose language is \textbf{too much}:

  \begin{itemize}
  \tightlist
  \item
    too verbose for the job
  \item
    too hard for the people who need to write it
  \end{itemize}
\item
  A DSL says \emph{exactly} what the task needs, in the task's own terms
\item
  The classic trade-off from ch 1: reuse an existing one\ldots{} or
  build your own when nothing fits
\end{itemize}

\note{Domain experts (here: water engineers) shouldn't need a CS degree
to express their problem.}
\end{frame}

\begin{frame}{The pipeline -- a teaser}
\phantomsection\label{the-pipeline-a-teaser}
\begin{itemize}
\item
  A language implementation transforms source text step by step
\item
  The three stages we build this semester:

  \textbf{Scanner} \(\rightarrow\) \textbf{Parser} \(\rightarrow\)
  \textbf{Interpreter}
\item
  Today, just the names. We build each in turn from Week 4.
\end{itemize}

\note{Ch 2's mountain metaphor: source text at the bottom, climb to
where the code's meaning is clear, descend to something the machine can
run. Our first interpreter is a \emph{tree-walk} interpreter -- climb
partway, then evaluate the tree directly.}
\end{frame}

\begin{frame}[fragile]{Source text becomes tokens}
\phantomsection\label{source-text-becomes-tokens}

\includegraphics[width=0.8\textwidth]{text/site/image/a-map-of-the-territory/tokens.png}

\emph{The scanner chunks characters into tokens.}

\note{Scanning (lexing): the linear stream of characters becomes
words/tokens. Whitespace and comments are discarded.}
\end{frame}

\begin{frame}{Tokens become a tree}
\phantomsection\label{tokens-become-a-tree}
\includegraphics[width=0.75\textwidth]{text/site/image/a-map-of-the-territory/ast.png}

\emph{The parser builds an abstract syntax tree from the flat token
stream.}

\note{Parsing gives the syntax a grammar -- the ability to compose
larger expressions from smaller parts, mirroring the nested structure of
the language.}
\end{frame}
